\section{背景}

\begin{frame}{Fan-out time 由哪三项构成？}
  {\footnotesize 一次 fan-out：父节点持有已提交 trunk，长度 $L$；harness 展开 $k$ 条 residual，长度 $\ell_i$；winner 绑定后只保留 spine，长度 $L{+}\ell_\star$，其余 abort。Fan-out time 是这一次操作从展开到提交的时间。\par}
  \vspace{0.2em}
  \why{只保留一条 spine。APC 在首次 fan-out 将 trunk 克隆 $k$ 份；DistServe 将已算出的页全部经 connector 传出；ESC / Speculative Rejection / DPTS 在 token 已经 resident 之后才打分。三项都发生在 abort 之前。}
  \begin{columns}[T]
    \begin{column}{0.32\textwidth}
      \begin{block}{Work：GPU prefill}
        \footnotesize 首次 fan-out 时，block 尚未进入 radix index。
        \[W = kL + \textstyle\sum_i \ell_i\]
      \end{block}
    \end{column}
    \begin{column}{0.32\textwidth}
      \begin{block}{Xfer：connector}
        \footnotesize 已 \texttt{insert}、随后 abort 的 residual 仍经该路径。$X$ 与 $W$ 为同一批页。
      \end{block}
    \end{column}
    \begin{column}{0.32\textwidth}
      \begin{block}{Decode-to-cut}
        \footnotesize 窗口、$\tau$ 与 mini-step 均在 prefill 之后，该段 decode 不能被剪去。
      \end{block}
    \end{column}
  \end{columns}
  % Citations kept out of the slides.
  % Yao, Yu, Zhao, Shafran, Griffiths, Cao, Narasimhan. Tree of Thoughts. NeurIPS 2023.
  % Kwon, Li, Zhuang, Sheng, Zheng, Yu, Gonzalez, Zhang, Stoica. PagedAttention. SOSP 2023.
  % Zhong, Liu, Chen, Hu, Zhu, Liu, Jin, Zhang. DistServe. OSDI 2024.

\section{设计}

\begin{frame}{\sys 的核心设计思想是哪两条？}
  \why{Prefill 与 decode 各用自己的分数和阈值。低分 residual 先 decode $t_0$，生成前缀达到阈值才续到 $D$，精度可回到 full fan-out。}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{1.\ P/D pruning 拆开}
        \footnotesize
        Prefill 读取 residual 文本，$\tau_{\mathrm{pre}}{=}0.15$ 只丢弃 loop。该决定改变 $W$，connector 只传输 $\mathrm{Keep}$。
        Decode 读取已生成前缀，阈值为 $0.45$，决定 probe 之后是否续到 $D$。
        两条权重不共享。Residual 上 illegal 为 $0.22$，生成前缀上仍为 illegal 的是 $0.16$。
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{2.\ 先 probe，再续到预算}
        \footnotesize
        Loop 在 kernel 之前停止。其余低于阈值的 residual 先 decode $t_0$。$s'_i\ge\tau$ 才从 $t_0$ 续到预算 $D$。
        MATH-500，$k{=}16$：整段丢弃 $s{<}\tau$ 为 \bad{$13/32$}、$52.1$\,s。
        $t_0{=}128$ 后续写为 \good{$20/32$}、\good{$72.2$\,s}。Full fan-out 为 $20/32$、$104.1$\,s。
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{一条规则如何同时决定 $W$ 和 $X$？}
  \why{每条 residual 只做一次 first-match。命中的动作同时确定 GPU prefill 的 token 与 connector 传输的 token。Index 0 只能是 HS、HP 或 PF，下一轮才有 spine。}

\end{frame}

\begin{frame}{五个动作如何映射到 $W$ 和 $X$？}
  \why{图上的 first-match 即 \texttt{\_decide} 的返回。整页命中为 HS / HP；未保留且前缀已失败为 EA，否则为 DS；其余为 PF。$X$ 只累加已保留且不是 HS 的 $W$，因此 EA 与 DS 不进入 connector。}
  {\scriptsize
  \begin{algorithmic}[1]
    \Function{decide}{$i$, prompt, keep}
      \State $m \gets \texttt{hash\_index.lookup}(\mathrm{prompt})$ \Comment{只认满页，$P{=}16$}
      \State $\mathrm{miss} \gets |\mathrm{prompt}|-m$ \textbf{if} $m{>}0$ \textbf{else} $n$
      \If{$m{>}0 \land \mathrm{miss}{=}0$}
        \State \Return HS, $W{=}0$, $X{=}0$ \Comment{整段已在 $H$}
      \ElsIf{$m{>}0 \land \mathrm{miss}{>}0 \land \mathrm{keep}$}
        \State \Return HP, $W{=}\mathrm{miss}$ \Comment{绿页不算}
      \ElsIf{$\neg\mathrm{keep} \land \texttt{early\_abort\_prefix}(\mathrm{text})$}
        \State \Return EA, $W{=}\lceil\alpha n\rceil$, $X{=}0$ \Comment{$\alpha{=}0.20$，不排队}
      \ElsIf{$\neg\mathrm{keep}$}
        \State \Return DS, $W{=}0$, $X{=}0$ \Comment{loop，或低分且前缀还没失败}
      \Else
        \State \Return PF, $W{=}n$
      \EndIf
    \EndFunction
    \State $X \gets W$ only if disagg, keep, and the action is HP or PF
    \State HS: 跳过。\quad DS、EA: \texttt{abort}，不 \texttt{queue}。\quad HP、PF: \texttt{queue\_known\_prefill}
  \end{algorithmic}}
  \src{\texttt{PrefillPruner.\_decide} 与 \texttt{queue\_fanout\_prefills}。\texttt{early\_abort\_prefix} 读文本前 $\alpha$，分数低于 $\tau_{\mathrm{pre}}$ 且 reason 不是 ok。Index 0 在 \texttt{rank} 里 \texttt{keep} 恒为真，所以到不了 EA / DS。低分即使有部分命中也不走 HP。}
\end{frame}

  \end{algorithmic}}

\section{结果}

\begin{frame}{Prefill pruning 相对三个 host 减少哪一项？}
  \why{Loop 的分数约 $0.02$，低于 $0.15$，在进入 kernel 之前丢弃。策略分数约 $1.0$，予以保留。Admission 不读取 host 的切点，因此三个 host 的 prefill 同时下降。ESC 的降幅最大，因其 baseline 将 loop decode 到窗口结束。}
  {\footnotesize
  \setlength{\tabcolsep}{5pt}
  \begin{tabular}{@{}lrrrr@{}}
    \toprule
    \textbf{$k{=}16$} & \textbf{prefill} & \textbf{e2e，单独} & \textbf{e2e，$+\tau_{\mathrm{pre}}{=}0.15$} & \textbf{any} \\
    \midrule

\begin{frame}{四个问题，各用一组数回答}
  \begin{enumerate}
    \item \textbf{为什么三篇的 prefill 一样？} 分数来自已经 resident 的 token。$k{=}16$ 时 ESC、SR、DPTS 的 prefill 都是 5008。加上 $\tau_{\mathrm{pre}}{=}0.15$ 后是 3848，ESC 从 17.05\,s 到 14.13\,s，仍是 $16/16$。
    \vspace{0.18em}
    \item \textbf{为什么不能共用一条 threshold？} Residual 上 illegal 是 $0.22$，生成前缀上仍是 illegal 是 $0.16$。MATH-500 $k{=}16$ 只用 $0.45$ 做 prefill 是 $13/32$、$52.1$\,s；$(0.15,0.45)$ 回到 $20/32$、$72.2$\,s。Full fan-out 是 $20/32$、$104.1$\,s。
    \vspace{0.18em}
    % \item \textbf{为什么默认是 $0.15$？} $k{=}16$ 上 $0.45$ 更快（ESC 10.92\,s，仍是 $16/16$），但 $k{=}4$ 和 $k{=}8$ 掉到 $14/16$。
    \item \textbf{我们的方法把效率提高了多少？} Game of 24 上 4B $1.61\to 1.13$\,s（$30\%$），8B $2.15\to 1.52$\,s（$29\%$）。GSM8K 8B 三轮 $2.48\to 1.91$\,s（$23\%$），只 prefill $48/192$。MATH-500 $k{=}16$ 的 probe $104.1\to 72.2$\,s（$31\%$），$20/32$ 不变。
    \vspace{0.18em}
    \item \textbf{Decoding、prefill、合并？} Decoding 不设 $\tau_{\mathrm{pre}}$，prefill 仍是 5008：ESC 预算 $D{=}512$ 是 17.05\,s，SR 切口 $\tau{=}256$ 是 13.14\,s，DPTS mini-step $100$ 是 11.86\,s。Prefill 只在 kernel 前剪，$\tau_{\mathrm{pre}}{=}0.15$ 是 14.13\,s，decode 侧再停 0。合并的 DPTS$+0.15$ 再停 4 条，是 11.49\,s。三轮 ToT 没接这三种算法：先 \texttt{queue\_fanout\_prefills}，live 孩子再 decode 32 token，只 prefill $48/192$。
  \end{enumerate}
