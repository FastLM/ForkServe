\section{背景}

\begin{frame}{一次 fan-out 的钱花在哪三步？}
  \why{只提交一条 spine，但 APC 把 trunk 克隆 $k$ 份，DistServe / Splitwise 把算出的页全部搬走，ESC / Speculative Rejection / DPTS 要等 token 已经在显存里才打分。}
  \begin{columns}[T]
    \begin{column}{0.32\textwidth}
      \begin{block}{Work，GPU prefill}
        \footnotesize 第一次 expand，block 还没进 radix index。
        \[W = kL + \textstyle\sum_i \ell_i\]
      \end{block}
    \end{column}
    \begin{column}{0.32\textwidth}
      \begin{block}{Xfer，connector}
        \footnotesize 被丢掉的 residual 如果已经 \texttt{insert}，也走这条链路。$X$ 与 $W$ 同一批页。
      \end{block}
    \end{column}
    \begin{column}{0.32\textwidth}
      \begin{block}{Decode-to-cut}
        \footnotesize 窗口、$\tau$ 或 mini-step 都在 prefill 之后。这段 decode 剪不掉。
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.35em}

\begin{frame}{ESC 要等到什么时候才停？}
  \why{熵为 0 读的是已经生成完的答案。图中窗口 $w{=}5$、计划样本 $L{=}20$。窗口关不上时，每个样本都已经 prefill，并 decode 到预算。}

\begin{frame}{Speculative Rejection 在砍分位之前已经付了什么？}
  \why{论文在 partial-reward horizon $\tau{=}256$ 才用 reward model 停掉较低分位 $\alpha$。停掉之前，每条候选的 prefill 和前 $\tau$ 个 token 已经付过。小 $k$ 上 memory-limit 触发器不响。}

\begin{frame}{DPTS 的 early-stop 读的是哪一段 token？}
  \why{每个孩子先生成固定 mini-step，置信度低于 $\theta_{\mathrm{es}}$ 才停。剪枝读的是已经生成的 mini-step。我们复现时这一步是 100 token。}

\begin{frame}{APC 的 block table 记下的是已经算出的页}
  \why{Scheduler 把请求交给 worker，KV cache manager 按页记账。命中发生在页已经存在之后。第一次 expand 时 trunk 还没有页，所以 $k$ 份 trunk 都要 prefill。}

\begin{frame}{RadixAttention 在请求处理之后才把前缀插进树}
  \why{图里的节点是这一步的输出写回去的。树一开始是空的。第一次 fan-out 的 $k$ 条 residual 都对不上已有节点。}

\end{frame}

\begin{frame}{DistServe 的箭头搬的是 prefiller 已经算出的 KV}
  \why{Controller 把请求先送给 prefill instance，再把 KV cache transfer 到 decoding instance。这条箭头上没有 admission。被丢掉的 residual 如果已经算完，也在这条链路上。}

\end{frame}

\begin{frame}{Splitwise 把 prompt 机的 KV 整段交给 token 机}
  \why{Prompt pool 算完 prompt，再经 KV-cache transfer 送到 token pool。搬运的是已经算出来的页，不是 admission 留下的 $\mathrm{Keep}$。}

\begin{frame}{一条规则怎么同时决定 $W$ 和 $X$？}
  \why{每条 residual 只做一次 first-match。命中的动作同时写下 GPU 要算的 token 和 connector 要搬的 token。Index 0 只能是 HS、HP 或 PF，下一轮才有 spine。}
  \paperfig[0.50]{fig_dragon.tex}
  \src{一格是 $P{=}16$ 个 token。绿页已经在 radix $H$ 里，橙页是这次 prefill，灰页 withheld。橙竖线是 HP 和 EA 的切口。}
\end{frame}

\begin{frame}{为什么是这五个动作，而且必须按这个顺序？}
  \why{先问「页是不是已经在 $H$ 里」，再问「文本是不是 loop」，最后才整段 prefill。若先判 PF，五条边都会变成整段橙页，账单回到 ESC / Speculative Rejection / DPTS 的 5008。}
  {\scriptsize
  \setlength{\tabcolsep}{3.5pt}
  \begin{tabular}{@{}lp{0.30\textwidth}p{0.22\textwidth}p{0.34\textwidth}@{}}
    \toprule
    & \textbf{什么时候} & \textbf{$W$ 和 $X$} & \textbf{为什么停在这里} \\
    \midrule
    HS & $m_i{=}\ell_i$，整段已在 $H$ & $W{=}X{=}0$ &
    页已经发布过。再 prefill 就是把同一条 residual 再算一遍，connector 也没有新页。 \\
    \addlinespace
    HP & $0{<}m_i{<}\ell_i$ & $W{=}\ell_i{-}m_i$；留下才 $X{=}W$ &
    绿页不重算。橙线右边才是新 KV。DistServe / Splitwise 会把绿页也搬走。 \\
    \addlinespace
    DS & $m_i{=}0$，$i{\in}\Lambda$，$i{\neq}0$ & $W{=}X{=}0$ &
    Loop 分数约 $0.02$，看文本就能认。连前缀都不算。连 loop 也 probe，同样 $20/32$，要 $74.4$\,s。 \\
    \addlinespace
    EA & $m_i{=}0$，不是 loop，前缀 $F_i{=}1$ & $W{=}\lceil\alpha\ell_i\rceil$，$X{=}0$ &
    前 $\alpha$ 已经低于 $\tau$。灰页不进 kernel，这段前缀也不 \texttt{insert}。 \\
    \addlinespace
    PF & $m_i{=}0$，不是 loop，$\neg F_i$ & $W{=}\ell_i$；留下才 $X{=}W$ &
    Index 0 和活策略走这里。Connector 搬的就是刚算出来的 miss，不是 trunk，也不是灰页。 \\
    \bottomrule
  \end{tabular}}
  \src{$m_i$ 按整页对齐，所以切口落在格边界上。$X{=}\sum_{i\in\mathrm{Keep},\,a_i\in\{\mathrm{HP},\mathrm{PF}\}}W_i$。控制面里这使 connector 从 11520 降到 32。}
\end{frame}

\begin{frame}{代码里，prefill 分数怎么打？}
  \why{Loop 记 $0.02$，低于默认 $0.15$，不进 kernel。Illegal text 记 $0.22$，默认留下。$0.45$ 才会把它一起丢掉。Index 0 永远留。}
  {\scriptsize
  \begin{algorithmic}[1]
    \Function{score\_text}{text}
      \If{text is blank}
        \State \Return $0$
      \ElsIf{an 8--40 character span repeats at least 4 times}
        \State \Return $0.02$ \Comment{loop}
      \ElsIf{\texttt{****}, unk, nan, or undefined}
        \State \Return $0.22$ \Comment{illegal}
      \Else
        \State \Return $0.55\min(1,2\alpha)+0.45\min(1,8\cdot\mathrm{entropy})$
      \EndIf
    \EndFunction
    \Function{rank}{residuals, $\tau_{\mathrm{pre}}{=}0.15$}
      \For{$i \gets 0$ \textbf{to} $k-1$}
        \State $\mathrm{keep}_i \gets (i{=}0) \lor \Call{score\_text}{\mathrm{text}_i}\ge\tau_{\mathrm{pre}}$
      \EndFor
    \EndFunction
  \end{algorithmic}}
  \src{\texttt{forkserve/prune.py}：\texttt{PrefillScorer.score\_text}，\texttt{BranchPruner.rank}。}
\end{frame}

\begin{frame}{五个动作在代码里怎么落到 $W$ 和 $X$？}
  \why{图上的 first-match 就是 \texttt{\_decide} 的返回。整页命中走 HS / HP；没留下、前缀已经失败走 EA，否则走 DS；剩下的才 PF。$X$ 只累加留下且不是 HS 的 $W$，所以 EA 和 DS 不进 connector。}
  {\scriptsize
  \begin{algorithmic}[1]
    \Function{decide}{$i$, prompt, keep}
      \State $m \gets \texttt{hash\_index.lookup}(\mathrm{prompt})$ \Comment{只认满页，$P{=}16$}
      \State $\mathrm{miss} \gets |\mathrm{prompt}|-m$ \textbf{if} $m{>}0$ \textbf{else} $n$
      \If{$m{>}0 \land \mathrm{miss}{=}0$}
        \State \Return HS, $W{=}0$, $X{=}0$ \Comment{整段已在 $H$}
      \ElsIf{$m{>}0 \land \mathrm{miss}{>}0 \land \mathrm{keep}$}
        \State \Return HP, $W{=}\mathrm{miss}$ \Comment{绿页不算}
      \ElsIf{$\neg\mathrm{keep} \land \texttt{early\_abort\_prefix}(\mathrm{text})$}
        \State \Return EA, $W{=}\lceil\alpha n\rceil$, $X{=}0$ \Comment{$\alpha{=}0.20$，不排队}
      \ElsIf{$\neg\mathrm{keep}$}
        \State \Return DS, $W{=}0$, $X{=}0$ \Comment{loop，或低分且前缀还没失败}
      \Else
        \State \Return PF, $W{=}n$
      \EndIf
    \EndFunction
    \State $X \gets W$ only if disagg, keep, and the action is HP or PF
    \State HS: 跳过。\quad DS、EA: \texttt{abort}，不 \texttt{queue}。\quad HP、PF: \texttt{queue\_known\_prefill}
  \end{algorithmic}}
  \src{\texttt{PrefillPruner.\_decide} 与 \texttt{queue\_fanout\_prefills}。\texttt{early\_abort\_prefix} 读文本前 $\alpha$，分数低于 $\tau_{\mathrm{pre}}$ 且 reason 不是 ok。Index 0 在 \texttt{rank} 里 \texttt{keep} 恒为真，所以到不了 EA / DS。低分即使有部分命中也不走 HP。}
\end{frame}

\begin{frame}{代码里，probe 和三轮 ToT 怎么接在一起？}
  \why{Loop 在 prefill 前跳过。其余低于 $\tau$ 的 residual 先 decode $t_0$，生成前缀过线才续到 $D$。每一轮都走 \texttt{queue\_fanout\_prefills}，只留 index 0。}
  \vspace{-0.25em}
  {\scriptsize
  \begin{algorithmic}[1]
    \Function{plan\_grow}{thoughts, $\tau$, $t_0$, $D$}
      \For{$i \gets 0$ \textbf{to} $k-1$}
        \If{$i\neq 0$ and $i$ is a repeated loop}
          \State skip \Comment{no prefill}
        \ElsIf{$i{=}0$ or $\mathrm{score}(\mathrm{thought}_i)\ge\tau$}
          \State prefill; decode to $D$
        \Else
          \State prefill; decode $t_0$; extend to $D$ iff $\mathrm{score\_prefix}\ge\tau$
        \EndIf
      \EndFor
    \EndFunction
    \For{$\mathrm{turn}\gets 0$ \textbf{to} $T-1$}
      \State thoughts $\gets$ \texttt{tot\_turn\_thoughts}(family, $k$, turn)
      \State fork each thought on the winner spine
      \State \texttt{queue\_fanout\_prefills} \Comment{the plan above}
      \If{turn is not the last}
        \State \texttt{generate\_nodes}(live children, step $=32$)
      \EndIf
      \State keep index 0; abort the rest
    \EndFor
    \State generate the answer on the committed spine
  \end{algorithmic}}
  \src{\texttt{experiments/prune\_ablation\_gpu.py} \texttt{plan\_grow}；\texttt{forkserve/bench.py} \texttt{run\_tot\_mt\_forest\_forkserve}。最后一轮的 step 是答案，不是 32 token。}
\end{frame}

\begin{frame}{三个问题，各用一组数回答}
  \begin{enumerate}
    \item \textbf{为什么三篇的 prefill 一样？} 分数来自已经 resident 的 token。$k{=}16$ 时 ESC、Speculative Rejection、DPTS 的 prefill 都是 5008。加上 $\tau_{\mathrm{pre}}{=}0.15$ 后是 3848，ESC 从 17.05\,s 到 14.13\,s，仍是 $16/16$。
    \vspace{0.55em}
    \item \textbf{为什么默认是 $0.15$？} $k{=}16$ 上 $0.45$ 更快（ESC 10.92\,s，仍是 $16/16$），但 $k{=}4$ 和 $k{=}8$ 掉到 $14/16$。MATH-500 上一刀切是 $13/32$，$t_0{=}128$ 回到 $20/32$，104.1\,s 到 72.2\,s。
    \vspace{0.55em}
    \item \textbf{哪些应用的端到端降了 $15\%$ 以上？} 降的是 prefill，不是把答案提前停掉。Game of 24 上 4B $1.61\to 1.13$\,s（$30\%$），8B $2.15\to 1.52$\,s（$29\%$）。GSM8K 8B 三轮从 2.48\,s 到 1.91\,s（$23\%$），只 prefill $48/192$。MATH-500 $k{=}16$ 的 probe 从 104.1\,s 到 72.2\,s（$31\%$），$20/32$ 不变。GSM8K ESC 在 $k{=}16$ 从 17.05\,s 到 10.92\,s（$\tau_{\mathrm{pre}}{=}0.45$）和 14.13\,s（$0.15$），都是 $16/16$。
  \end{enumerate}
\end{frame}
