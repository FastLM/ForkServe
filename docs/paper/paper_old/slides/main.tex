\section{背景}

\begin{frame}{一次 fan-out 的三笔账}
  Agent 和 Tree-of-Thoughts 的服务单位是一次分支：共享一条 trunk，分叉出 $k$ 条短 residual，最后只提交一条 spine。传统栈对这次分支收三笔钱。
  \vspace{0.2em}
  \begin{columns}[T]
    \begin{column}{0.32\textwidth}
      \begin{block}{Work，GPU prefill}
        \footnotesize 第一次 expand 时 block 还没进 radix index。APC 和 hash prefill 把 trunk 克隆 $k$ 份。
        \[W = kL + \textstyle\sum_i \ell_i\]
      \end{block}
    \end{column}
    \begin{column}{0.32\textwidth}
      \begin{block}{Xfer，connector}
        \footnotesize DistServe / Splitwise 把 prefiller 算出的每一页传给 decoder。被丢掉的 residual 如果已经 insert，也在这条链路上。
      \end{block}
    \end{column}
    \begin{column}{0.32\textwidth}
      \begin{block}{Decode-to-cut}
        \footnotesize ESC、Speculative Rejection、DPTS 的分数来自已经 resident 的 token。Prefill 和一段 decode 发生在丢弃之前。
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.15em}
  \begin{exampleblock}{后面的设计对着后两笔账}
    \footnotesize Admission 在 kernel 之前丢掉不该留的 residual。Connector 只搬 $\mathrm{Keep}$ 的 miss。Decode-time pruning 剪不到这两笔。
  \end{exampleblock}
  \vspace{0.2em}
  {\fontsize{6.2}{7.0}\selectfont
  Yao, Yu, Zhao, Shafran, Griffiths, Cao, Narasimhan. \textit{Tree of Thoughts: Deliberate Problem Solving with Large Language Models.} NeurIPS 2023.\\
  Kwon, Li, Zhuang, Sheng, Zheng, Yu, Gonzalez, Zhang, Stoica. \textit{Efficient Memory Management for Large Language Model Serving with PagedAttention.} SOSP 2023.\\
  Zhong, Liu, Chen, Hu, Zhu, Liu, Jin, Zhang. \textit{DistServe: Disaggregating Prefill and Decoding for Goodput-optimized Large Language Model Serving.} OSDI 2024.\\
  Patel, Choukse, Zhang, Shah, Goiri, Maleki, Bianchini. \textit{Splitwise: Efficient Generative LLM Inference Using Phase Splitting.} ISCA 2024.
  }
\end{frame}

\begin{frame}{Speculative Rejection：部分奖励到了 $\tau$ 再砍分位}
  {\scriptsize Sun, Haider, Zhang, Yang, Qiu, Yin, Wang, Bartlett, Zanette. \textit{Fast Best-of-N Decoding via Speculative Rejection.} NeurIPS 2024.}
  \vspace{0.2em}

  \end{center}
  \vspace{0.1em}
  {\small
  Best-of-$N$ 把 $N$ 条生成全部跑完，再用 reward model 挑最高分。Speculative Rejection 先用很大的 batch 开跑，在 partial-reward horizon $\tau$（论文里 $\tau{=}256$）用 reward model 给半成品打分，停掉较低分位 $\alpha$，把省下的显存让给还活着的候选。
  停掉之前，每条候选的 prefill 和前 $\tau$ 个 token 已经付过。Memory-limit 触发器在小 $k$ 上不会响。}
  \src{图：原文 Figure 1。Sun, Haider, Zhang, Yang, Qiu, Yin, Wang, Bartlett, Zanette. \textit{Fast Best-of-N Decoding via Speculative Rejection.} NeurIPS 2024.}

\begin{frame}{五个动作：同一条规则决定 $W$ 和 $X$}
  {\small 先读 radix 命中 $m_i$。整段命中不进 kernel；部分命中只算 miss；重复 loop 直接丢掉；其余低分只探前缀；过线的才整段 prefill。Index 0 只能是 HS、HP 或 PF。}

\end{frame}

\section{结果}

\begin{frame}{同一个 $\tau_{\mathrm{pre}}$ 插在三个 host 前面}
  {\small $k{=}16$ 上 $0.15$ 和 $0.45$ 都是 $16/16$。端到端少多少，取决于这个 host 本来会把 loop decode 多久。$k{=}8$ 时 $0.15$ 把 SR 和 DPTS 从 $15/16$ 抬回 $16/16$。}
  \vspace{0.15em}
  \paperfig[0.62]{fig_eval_admit.tex}
  \src{论文图。GSM8K，Qwen3-8B，$n{=}16$，$D{=}512$。来源 \texttt{logs/prune\_ablation\_thresh/all.json}。}
\end{frame}

\begin{frame}{为什么不能把低分全部挡在 prefill 前}
  {\small MATH-500，$n{=}32$。Loop 是 0 分，可以在 kernel 前丢掉。Illegal text 要先探 $t_0$。$t_0{=}128$ 在 $k{=}16$ 回到 $20/32$。}
  \vspace{0.15em}
  \paperfig[0.62]{fig_eval_grow.tex}
  \src{论文图。Grow 行与 2026-10-01 的 shard JSON 一致。Full / 一刀切来自同一次实验的 tab:grow。}
\end{frame}

\begin{frame}{P/D 分离之后，pruning 少搬了什么}
  DistServe 和 Splitwise 的 connector 搬运 prefiller 已经算出的页。Admission 把 $\mathrm{Keep}$ 定在 \texttt{insert} 之前。

  Kwon, Li, Zhuang, Sheng, Zheng, Yu, Gonzalez, Zhang, Stoica. \textit{Efficient Memory Management for Large Language Model Serving with PagedAttention.} SOSP 2023.}
  \vspace{0.3em}
  {\footnotesize
  \setlength{\tabcolsep}{5pt}
  \begin{tabular}{@{}lrrrr@{}}
    \toprule
    & \textbf{APC} & \textbf{P/D} & \textbf{\sys} & \textbf{\fsplus} \\

\begin{frame}{三句话}
  \begin{enumerate}
    \item ESC 等一个窗口的完整样本，Speculative Rejection 等到 $\tau$，DPTS 等到 mini-step。三篇的分数都来自已经 resident 的 token，所以 $k{=}16$ 时 computed prefill 都是 5008。P/D 接着把这些页全部传走。
    \vspace{0.4em}
    \item 这里的 admission 在 kernel 之前处理重复 loop，其他低分只给 $t_0$ 的 probe。Connector 只搬 $\mathrm{Keep}$ 的 miss。五个动作同时决定 $W$ 和 $X$。
    \vspace{0.4em}
    \item 最好的安全点是 $\tau_{\mathrm{pre}}{=}0.15$。宽树上 $\tau_{\mathrm{pre}}{=}0.45$ 仍是 $16/16$，ESC 从 17.05\,s 降到 10.92\,s。MATH-500 上 $t_0{=}128$ 把 $20/32$ 的精度留住，时间从 104.1\,s 降到 72.2\,s。
  \end{enumerate}
\end{frame}
