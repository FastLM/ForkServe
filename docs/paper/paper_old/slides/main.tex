\section{背景}

\begin{frame}{一次 fan-out 的三笔账}
  Agent 和 Tree-of-Thoughts 的服务单位是一次分支：共享一条 trunk，分叉出 $k$ 条短 residual，最后只提交一条 spine。传统栈对这次分支收三笔钱。
  \vspace{0.6em}
  \begin{columns}[T]
    \begin{column}{0.32\textwidth}
      \begin{block}{Work，GPU prefill}
        \small 第一次 expand 时 block 还没进 radix index。APC 和 hash prefill 把 trunk 克隆 $k$ 份。
        \[W = kL + \textstyle\sum_i \ell_i\]
      \end{block}
    \end{column}
    \begin{column}{0.32\textwidth}
      \begin{block}{Xfer，connector}
        \small DistServe / Splitwise 把 prefiller 算出的每一页传给 decoder。Aborted residual 也在这条链路上。
      \end{block}
    \end{column}
    \begin{column}{0.32\textwidth}
      \begin{block}{Decode-to-cut}
        \small ESC、Speculative Rejection、DPTS 的分数来自已经 resident 的 token。Prefill 和一段 decode 发生在丢弃之前。
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.5em}
  \begin{exampleblock}{后面的设计对着这三笔账}
    \small Alias 去掉 $W$ 和 $X$ 里的 $L$。Admission 把不该留的 residual 挡在 kernel 和 connector 之前。Abort 之后活着的 \kv{} 只剩 spine。
  \end{exampleblock}
\end{frame}

\begin{frame}{Tree of Thoughts 具体怎么搜}
  \small Yao, Yu, Zhao, Shafran, Griffiths, Cao, Narasimhan.\ \textit{Tree of Thoughts: Deliberate Problem Solving with Large Language Models.} NeurIPS 2023.
  \vspace{0.45em}
  {\footnotesize
  \setlength{\tabcolsep}{4pt}
  \begin{tabular}{@{}p{0.13\textwidth}p{0.42\textwidth}p{0.36\textwidth}@{}}
    \toprule
    \textbf{步骤} & \textbf{具体做法} & \textbf{对 serving 的含义} \\
    \midrule
    Thought &
    中间状态是一段推理。从当前 state 采样 $k$ 个 candidate thought。 &
    $k$ 条 residual 共享父 prompt，分叉只在短后缀。 \\
    \addlinespace
    Value &
    对已经生成完的 thought 打分：value prompt，或多次投票。 &
    分数读的是已经 decode 出来的文本。 \\
    \addlinespace
    Search &
    BFS 保留 top-$b$ beam；DFS 先走最高分，低于阈值就剪。 &
    剪枝发生在这一步生成结束之后。 \\
    \addlinespace
    下一层 &
    被留下的 beam 各自成为下一层的父节点，再展开。 &
    每一层都要把 $k$ 份上下文重新交给引擎。 \\
    \bottomrule
  \end{tabular}}
  \vspace{0.55em}
  ToT 把搜索单位从 token 抬到 thought。引擎侧仍是 $k$ 次独立 generate。Beam 被丢掉时，这条 thought 的 prefill 和这一步的 decode 已经付过。
  \src{同方向：Hive（Luo et al., arXiv 2026）在算法层和任务层扩 branching，\kv{} 仍按请求隔离。}
\end{frame}

\begin{frame}{三种 decode-time pruning，各自停在哪}
  三条线都在模型已经写出一段 token 之后才决定丢掉谁。差别是这段前缀有多长。
  \vspace{0.4em}
  {\scriptsize
  \setlength{\tabcolsep}{3.5pt}
  \begin{tabular}{@{}p{0.18\textwidth}p{0.22\textwidth}p{0.30\textwidth}p{0.22\textwidth}@{}}
    \toprule
    \textbf{方法} & \textbf{出处} & \textbf{具体规则} & \textbf{丢掉之前必须生成} \\
    \midrule
    ToT beam &
    Yao et al., NeurIPS 2023 &
    一步 thought 生成完再打分，留 top beam。 &
    这一步的完整 decode \\
    \addlinespace
    ESC &
    Li et al., ICLR 2024 &
    Self-consistency 抽多个完整样本。窗口 $w$ 内答案一致，才跳过后面的样本。 &
    窗口里每个样本的全程 decode \\
    \addlinespace
    Speculative Rejection &
    Sun et al., NeurIPS 2024 &
    Best-of-$N$。生成到 partial-reward horizon $\tau$，按部分奖励排序，停掉较低分位 $\alpha$。 &
    每个候选的 prefill $+\,\tau$ 个 token \\
    \addlinespace
    DPTS &
    Ding et al., ACL 2025 &
    每个活跃节点先生成固定 mini-step，再按置信度 early-stop 低分孩子。 &
    每个节点的 prefill $+$ mini-step \\
    \bottomrule
  \end{tabular}}
  \vspace{0.45em}
  \begin{alertblock}{在我们的 fan-out 上，窗口经常关不上}
    \small ESC 的窗口是 5 个一致样本。$k{=}4$ 且有两条低分时，窗口一直开着，四个孩子都 decode 到预算。Speculative Rejection 的 memory-limit 触发器在 $k{=}4$ 不会响，更早的检查点是 $\tau{=}256$。DPTS 最早，mini-step 仍是 100 token。
  \end{alertblock}
\end{frame}

\begin{frame}{Prefix cache 与 P/D 分离具体做什么}
  {\scriptsize
  \setlength{\tabcolsep}{3.2pt}
  \begin{tabular}{@{}p{0.16\textwidth}p{0.20\textwidth}p{0.34\textwidth}p{0.24\textwidth}@{}}
    \toprule
    \textbf{系统} & \textbf{出处} & \textbf{机制} & \textbf{第一次 fan-out} \\
    \midrule
    Orca & Yu et al., OSDI 2022 &
    Iteration-level scheduling，prefill 与 decode 混在一个 batch。 &
    仍按请求算完整 prompt。 \\
    \addlinespace
    vLLM APC & Kwon et al., SOSP 2023 &
    PagedAttention。整块 \kv{} 用 $\mathrm{hash}(\mathrm{parent},\mathrm{tokens},\mathrm{extra})$ 做 automatic prefix caching。 &
    块必须先存在才能命中。同 batch 的 $k$ 条分支全部 miss，克隆 $k$ 份 trunk。 \\
    \addlinespace
    SGLang radix & Zheng et al., NeurIPS 2024 &
    RadixAttention 用 token 前缀做 radix tree。 &
    共享是在 token 写出来之后发现的。 \\
    \addlinespace
    Sarathi-Serve & Agrawal et al., OSDI 2024 &
    Chunked prefill，把 prefill 切块塞进 decode 的空闲算力。 &
    减少的是气泡，不是分支份数。 \\
    \addlinespace
    DistServe & Zhong et al., OSDI 2024 &
    Prefill 实例和 decode 实例拆开，按 goodput 放置，中间传 \kv{}。 &
    传的是 prefiller 算出的每一页。 \\
    \addlinespace
    Splitwise & Patel et al., ISCA 2024 &
    Prompt phase 和 token phase 分到不同机器。 &
    $X = kL + \sum_i \ell_i$，和第一次 expand 的 $W$ 一样大。 \\
    \bottomrule
  \end{tabular}}
  \src{赢家之后再开一次请求，APC 可以 hash-hit。第一次 expand 本身省不下来。}
\end{frame}

\begin{frame}{三条线停在同一个地方}
  共享 trunk 的系统等块写完才发现共享。丢弃分支的系统等 token 写完才打分。P/D 把这两件事的产物整页搬走。
  \vspace{0.35em}
  {\scriptsize
  \setlength{\tabcolsep}{3.5pt}
  \begin{tabular}{@{}p{0.24\textwidth}p{0.18\textwidth}p{0.28\textwidth}p{0.22\textwidth}@{}}
    \toprule
    \textbf{方法} & \textbf{第一次 expand} & \textbf{最早能丢掉非赢家} & \textbf{Connector} \\
    \midrule
    Recompute & $k$ 份 $L$ & 请求结束 & 无 \\
    APC / radix / hash prefill & $k$ 份 $L$ & 块发布之后的 LRU & 无；赢家重放才是 hit \\
    P/D disaggregation & $k$ 份 $L$ & insert 之后 & $kL+\sum_i\ell_i$ \\
    ESC / SR / DPTS & 每条 residual & 一段已生成前缀之后 & 每个已 prefill 孩子的 \kv{} \\
    \good{\sys} & \good{$L$ 的一份 alias} & \good{loop 在 kernel 前；其余低分在 $t_0$ probe 后} & \good{$\mathrm{Keep}$ 的 miss tail} \\
    \bottomrule
  \end{tabular}}
  \vspace{0.45em}
  \begin{block}{占用也是三档}
    \small
    $M_{\mathrm{clone}}=b(kL+\sum_i\ell_i)$，\
    $M_{\mathrm{CoW}}=b(L+\sum_i\ell_i)$，\
    $M_{\mathrm{spine}}=b(L+\ell_\star)$。
    APC、recompute、baseline disaggregation 的第一次 expand 落在 clone。Abort 之后该留下的是 spine。
  \end{block}
\end{frame}

\section{设计}

\begin{frame}{四条核心设计}
  \sys{} 坐在 agent runtime 和 LLM engine 之间。Harness 交出分支结构；控制面负责 admit、prune、bind、retain；backend 搬 page 和 token。用户可见 token 的关键路径上只有 \texttt{commit}。
  \vspace{0.5em}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{1\quad CoW context tree}
        \small \texttt{fork} 只写两个字：parent 指针和 \texttt{alias\_len}，并对被 alias 的 frame 做 incref。$O(1)$ 于 $L$。Abort 只还 residual 的 page。
      \end{block}
      \vspace{0.35em}
      \begin{block}{3\quad Probe，而不是一刀切}
        \small 重复 loop 在 prefill 前丢掉。其他低分先 decode $t_0$ 个 token，probe 分数过 $\tau$ 才继续到预算 $D$。Index 0 永远留在 $\mathrm{Keep}$。
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{2\quad 五动作 admission}
        \small 先读 radix 命中 $m_i$，再读是不是重复 loop。一次 first-match 同时定下 $W_i$ 和 $X_i$。Trunk 的 $L$ 两边都不出现。
      \end{block}
      \vspace{0.35em}
      \begin{block}{4\quad Connector 与调度看同一个 Keep}
        \small $X$ 只加 $\mathrm{Keep}$ 里真正 miss 的尾巴。Abort 后池子按 spine 计并发。Chunk 预算先给 committed decode，剩下的才去 prefill 下一次 miss。
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Copy-on-write context tree}
  一个 session 是一棵有根树。节点保存 token 序列、相对父节点的 residual、模式（Spec / Commit / Idle / Dead），以及逻辑页表：parent、\texttt{alias\_len}、自己的 page。
  \vspace{0.15em}
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      \textbf{不变量}
      \begin{itemize}\setlength{\itemsep}{0.2em}
        \item 父序列是子序列的前缀。
        \item 除 join 期间，一个节点最多一个 committed child。
        \item Speculative 节点对 sampler 不可见。
        \item 一页的 refcount 等于还映射着它的非 dead 节点数。
      \end{itemize}
      {\small 多层 fan-out 是一串 CoW。活路径共享公共前缀上的每一页。没有活孩子的子树级联 abort，停在仍有活孩子的节点。}
    \end{column}
    \begin{column}{0.52\textwidth}
      \textbf{API。\texttt{commit} 才在关键路径上}
      \vspace{0.25em}
      {\scriptsize
      \setlength{\tabcolsep}{4pt}
      \begin{tabular}{@{}lp{0.34\textwidth}@{}}
        \toprule
        \textbf{调用} & \textbf{作用} \\
        \midrule
        \texttt{open(session)} & 建根，prefill system + user \\
        \texttt{fork(u, bid, tok?)} & $u$ 的孩子；可选已知后缀 \\
        \texttt{speculate(v, pri)} & 标记 $v$ 可以投机 prefill \\
        \texttt{commit(u, tokens)} & 按 LCP 绑定，开始 decode \\
        \texttt{join(children, pol)} & all / first / k-of-n \\
        \texttt{abort(v)} / \texttt{close} & 丢掉子树 / 结束 session \\
        \bottomrule
      \end{tabular}}
    \end{column}
  \end{columns}
  \vspace{0.35em}
  \begin{exampleblock}{Abort 还的是 residual}
    \small 释放 $b\cdot\ell_i$，而不是 $b\cdot(L+\ell_i)$。LCP commit 把赢家的整页发布进 APC index，后面的 session 还能 hash-hit。Spec page 自己不进 index。
  \end{exampleblock}
\end{frame}

\begin{frame}{一个 first-match 动作，同时决定 $W$ 和 $X$}
  $m_i$ 是 residual 里已经在 radix index $H$ 中、按页对齐的最长前缀。读 $m_i$ 不花模型 FLOPs。只有 miss 才看这条是不是重复 loop。
  \vspace{0.35em}
  {\footnotesize
  \setlength{\tabcolsep}{4pt}
  \begin{tabular}{@{}p{0.20\textwidth}p{0.32\textwidth}p{0.16\textwidth}p{0.24\textwidth}@{}}
    \toprule
    \textbf{动作} & \textbf{条件} & \textbf{Work} & \textbf{Xfer} \\
    \midrule
    HS，hash skip & $m_i=\ell_i$，整段命中 & $0$ & $0$ \\
    HP，hash partial & $0<m_i<\ell_i$ & $\ell_i-m_i$ & 留在 Keep 才搬 miss \\
    \good{DS，draft skip} & \good{$m_i=0$ 且 $i\in\Lambda$（重复 loop）} & \good{$0$} & \good{$0$} \\
    EA，early abort & miss、不是 loop、前 $\alpha$ 段分数 $<\tau$ & $\lceil\alpha\ell_i\rceil$ & $0$ \\
    PF，prefill & miss、不是 loop、分数过线 & $\ell_i$ & 留在 Keep 才搬 \\
    \bottomrule
  \end{tabular}}
  \vspace{0.5em}
  Index 0 强制进 $\mathrm{Keep}$，所以 $a_0\in\{\mathrm{HS},\mathrm{HP},\mathrm{PF}\}$。Alias 已经把 trunk 指到父页表，所以 $L\notin W_i$，也 $L\notin X_i$。Baseline disaggregation 搬的是 $kL+\sum_i\ell_i$。
  \src{冷 index 上 $W_{\mathrm{APP}}\le W_{\mathrm{CoW}}\le W_{\mathrm{APC}}=W_{\mathrm{PD}}$。有一条 loop 被 draft-skip，则 $W$ 严格更小；$\mathrm{Keep}\neq[k]$ 或有一条命中，则 $X$ 严格小于 PD。}
\end{frame}

\begin{frame}{Loop 直接丢，其余低分只探 $t_0$}
  把所有 $s_i<\tau$ 都挡在 prefill 前，会把还能做对的 illegal text 一起丢掉。Probe admission 收成三条。
  \vspace{0.35em}
  {\footnotesize
  \setlength{\tabcolsep}{4pt}
  \begin{tabular}{@{}p{0.24\textwidth}p{0.16\textwidth}p{0.24\textwidth}p{0.28\textwidth}@{}}
    \toprule
    \textbf{分支} & \textbf{Prefill 分数} & \textbf{动作} & \textbf{Decode 长度} \\
    \midrule
    $i\in\Lambda$，重复 loop & 约 $0.02$ & DS，kernel 之前丢掉 & $0$ \\
    \addlinespace
    $i=0$，或 $s_i\ge\tau$ & 策略分支约 $1.0$ & PF & 预算 $D$ \\
    \addlinespace
    其他低分 & illegal text 约 $0.22$ & 先 PF，再 decode $t_0$ & $s'_i\ge\tau$ 则续到 $D$，否则停在 $t_0$ \\
    \bottomrule
  \end{tabular}}
  \vspace{0.45em}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{两个阈值不是一回事}
        \small Prefill 阈值默认 $\tau_{\mathrm{pre}}=0.15$，只掉 loop。Decode probe 在 MATH-500 上是 $\tau=0.45$，$t_0\in\{64,128\}$。Host 的 $\tau$、DPTS step、$\alpha$ 都不参与 prefill admission。
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{剩余算力留给下一次 miss}
        \small Abort 之后 $M=b(L+\ell_\star)$，并发槽位是 $\lfloor H/M\rfloor$。Chunk 预算先满足 committed decode，剩下的 $B^{s}$ 去 prefill 下一条 miss，或在 idle 够长时 prefill 已知后缀。投机 token 不进 sampler。
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\section{结果}

\begin{frame}{Prefill 剪枝比 decode 剪枝少算了什么}
  Qwen3-8B，GSM8K，$n{=}16$，$D{=}512$，$\mathrm{tp}{=}1$，四张 A100-40GB。$k{=}16$ 时三个 host 单独跑，computed prefill 都是 5008。Decode 阶段的剪枝没有少掉任何一次 residual prefill。
  \vspace{0.4em}
  {\footnotesize
  \setlength{\tabcolsep}{5pt}
  \begin{tabular}{@{}lrrr@{}}
    \toprule
    & \textbf{Host 单独} & \textbf{$+\tau_{\mathrm{pre}}{=}0.15$} & \textbf{变化} \\
    \midrule
    Admitted residual & 16 & \good{12} & \good{$-4$} \\
    Computed prefill & 5008 & \good{3848} & \good{$-23\%$} \\
    ESC decode tokens & 131072 & 98304 & $-25\%$ \\
    ESC 端到端 & 17.05\,s & \good{14.13\,s} & \good{$-17\%$} \\
    \bottomrule
  \end{tabular}}
  \vspace{0.4em}
  {\small 代价模型把同一笔账算得更极端：ESC 要 decode 20480 token 才关窗（276\,ms），DPTS 的 100-token mini-step 仍在被丢掉的孩子上花 2000 token（78.7\,ms）。\fsplus{} 在那些 residual 上的 decode 是 0，fan-out 31.4\,ms。GPU 上，被 admission 丢掉的孩子不再占 prefill，也不再占 decode slot。}
  \src{来源：\texttt{logs/prune\_ablation\_thresh/all.json}，2026-10-01。端到端 $=$ trunk warm $+$ fan-out。三条 host 的 prefill token 相同。}
\end{frame}

\begin{frame}{同一个阈值插在三个 host 前面}

\begin{frame}{为什么不能把低分全部挡在 prefill 前}
  MATH-500 level $\ge 4$，Qwen3-8B，$n{=}32$，$D{=}1024$，$\tau{=}0.45$。一半 residual 是竞赛策略，另一半在重复 loop 和 illegal text 之间交替。Any-finished：某条跑满预算的分支对上参考答案。

\begin{frame}{三句话}
  \begin{enumerate}
    \item ToT 的 beam、ESC 的一致窗口、Speculative Rejection 的 $\tau$、DPTS 的 mini-step，分数都来自已经 resident 的 token。DistServe / Splitwise 接着把这些页全部传走。APC 要等块写完才发现 trunk 是共享的。
    \vspace{0.45em}
    \item \sys{} 用一次 alias 去掉 $L$，用一次 first-match 决定这条 residual 是命中、部分命中、loop、提前停、还是 prefill。Loop 在 kernel 前丢掉；其他低分只给 $t_0$ 的 probe。Connector 只搬 $\mathrm{Keep}$ 的 miss。
    \vspace{0.45em}
    \item GPU 上，$\tau_{\mathrm{pre}}{=}0.15$ 把 $k{=}16$ 的 computed prefill 从 5008 降到 3848，any-finished 保持 $16/16$，ESC 从 17.05\,s 降到 14.13\,s。三轮 ToT 里 192 条 residual 只 prefill 48 条，expand 是 APC 的 $7.9$--$10.6$ 倍，peak \kv{} 是 spine。
  \end{enumerate}
  \vspace{0.6em}
  {\scriptsize\color{fsgray}
  数字来自 \texttt{logs/prune\_ablation\_\{thresh,grow,wide\}}（2026-10-01）和 \texttt{logs/tot\_mt}（2026-09-29）。
  P/D 的 139 / 162 / 47 / 31.4\,ms 是控制面代价模型。
  MATH-500 的 full fan-out 与“丢掉全部 $s{<}\tau$”来自论文 tab:grow，grow 行已用 shard JSON 核对。}
\end{frame}
