\section{背景}

\begin{frame}{一次 fan-out 的钱花在哪三步？}
  \why{只提交一条 spine，但 APC 把 trunk 克隆 $k$ 份，DistServe 把算出的页全部搬走，ESC / Speculative Rejection / DPTS 要等 token 已经在显存里才打分。}
  \begin{columns}[T]
    \begin{column}{0.32\textwidth}
      \begin{block}{Work，GPU prefill}
        \footnotesize 第一次 expand，block 还没进 radix index。
        \[W = kL + \textstyle\sum_i \ell_i\]
      \end{block}
    \end{column}
    \begin{column}{0.32\textwidth}
      \begin{block}{Xfer，connector}
        \footnotesize 被丢掉的 residual 如果已经 \texttt{insert}，也走这条链路。$X$ 与 $W$ 同一批页。
      \end{block}
    \end{column}
    \begin{column}{0.32\textwidth}
      \begin{block}{Decode-to-cut}
        \footnotesize 窗口、$\tau$ 或 mini-step 都在 prefill 之后。这段 decode 剪不掉。
      \end{block}
    \end{column}
  \end{columns}

\section{设计}

\begin{frame}{\sys 的核心设计思想是哪两条？}
  \why{Prefill 和 decode 各用自己的分数、自己的线。低分 residual 先长出 $t_0$，前缀过线才续到 $D$，剪完精度还能回到 full fan-out。}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{1.\ P/D pruning 拆开}
        \footnotesize
        Prefill 读 residual 文本，$\tau_{\mathrm{pre}}{=}0.15$ 只丢 loop。这一刀改 $W$，connector 只搬 $\mathrm{Keep}$。
        Decode 读已经生成的前缀，线是 $0.45$，决定 probe 之后续不续到 $D$。
        两条权重不共享。Residual 上 illegal 是 $0.22$，生成前缀上仍是 illegal 是 $0.16$。
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{2.\ 先 probe，再长回去}
        \footnotesize
        Loop 停在 kernel 前。其余低于线的 residual 先 decode $t_0$。$s'_i\ge\tau$ 才从 $t_0$ 续到预算 $D$。
        MATH-500，$k{=}16$：整段丢掉 $s{<}\tau$ 是 \bad{$13/32$}、$52.1$\,s。
        $t_0{=}128$ 长回去是 \good{$20/32$}、\good{$72.2$\,s}。Full fan-out 是 $20/32$、$104.1$\,s。
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{一条规则怎么同时决定 $W$ 和 $X$？}
  \why{每条 residual 只做一次 first-match。命中的动作同时写下 GPU 要算的 token 和 connector 要搬的 token。Index 0 只能是 HS、HP 或 PF，下一轮才有 spine。}
  \paperfig[0.50]{fig_dragon.tex}
  \src{一格是 $P{=}16$ 个 token。绿页已经在 radix $H$ 里，橙页是这次 prefill，灰页 withheld。橙竖线是 HP 和 EA 的切口。}
\end{frame}

\begin{frame}{上一轮刷新、这一轮分数、LRU，为什么和 decode 是两套？}
  \why{页留不留，只看三件事：上一轮把赢家的满页发布进 hash，还被引用的页 touch 到 LRU 外面；这一轮用 prefill 分数决定 residual 进不进 kernel，命中的满页再 touch 一次；要新页时从 free queue 头部拿，头部若还挂着 hash 就先摘掉。Decode 的分数读的是已经生成的前缀，不发布页，也不动这条队列。}

  \setlength{\tabcolsep}{5pt}
  \begin{tabular}{@{}lccc@{}}
    \toprule
    & \textbf{上一轮 request} & \textbf{这一轮 request} & \textbf{LRU} \\
    \midrule
    cache & 满页 \texttt{publish} & \texttt{lookup} 只认满页 & 空闲页挂在队尾 \\
    & 还在用的页 \texttt{touch} & 命中再 \texttt{touch} 一次 & 分配从队头摘 hash \\
    \midrule
    分数 & 不打 & prefill，$\tau_{\mathrm{pre}}{=}0.15$ & 不读 \\
    \bottomrule
  \end{tabular}}
  \vspace{0.45em}
  {\footnotesize
  \setlength{\tabcolsep}{6pt}
  \begin{tabular}{@{}lccc@{}}
    \toprule
    & \textbf{读的文本} & \textbf{线} & \textbf{决定什么} \\
    \midrule
    prefill，$a$ & 这一轮的 residual，kernel 之前 & $a{=}0.15$ & 进不进 kernel，改不改 $W$ \\
    decode，$b$ & 已经生成的前缀 & $b{=}0.45$ & probe 之后续不续到 $D$ \\

\begin{frame}{五个动作在代码里怎么落到 $W$ 和 $X$？}
  \why{图上的 first-match 就是 \texttt{\_decide} 的返回。整页命中走 HS / HP；没留下、前缀已经失败走 EA，否则走 DS；剩下的才 PF。$X$ 只累加留下且不是 HS 的 $W$，所以 EA 和 DS 不进 connector。}
  {\scriptsize
  \begin{algorithmic}[1]
    \Function{decide}{$i$, prompt, keep}
      \State $m \gets \texttt{hash\_index.lookup}(\mathrm{prompt})$ \Comment{只认满页，$P{=}16$}
      \State $\mathrm{miss} \gets |\mathrm{prompt}|-m$ \textbf{if} $m{>}0$ \textbf{else} $n$
      \If{$m{>}0 \land \mathrm{miss}{=}0$}
        \State \Return HS, $W{=}0$, $X{=}0$ \Comment{整段已在 $H$}
      \ElsIf{$m{>}0 \land \mathrm{miss}{>}0 \land \mathrm{keep}$}
        \State \Return HP, $W{=}\mathrm{miss}$ \Comment{绿页不算}
      \ElsIf{$\neg\mathrm{keep} \land \texttt{early\_abort\_prefix}(\mathrm{text})$}
        \State \Return EA, $W{=}\lceil\alpha n\rceil$, $X{=}0$ \Comment{$\alpha{=}0.20$，不排队}
      \ElsIf{$\neg\mathrm{keep}$}
        \State \Return DS, $W{=}0$, $X{=}0$ \Comment{loop，或低分且前缀还没失败}
      \Else
        \State \Return PF, $W{=}n$
      \EndIf
    \EndFunction
    \State $X \gets W$ only if disagg, keep, and the action is HP or PF
    \State HS: 跳过。\quad DS、EA: \texttt{abort}，不 \texttt{queue}。\quad HP、PF: \texttt{queue\_known\_prefill}
  \end{algorithmic}}
  \src{\texttt{PrefillPruner.\_decide} 与 \texttt{queue\_fanout\_prefills}。\texttt{early\_abort\_prefix} 读文本前 $\alpha$，分数低于 $\tau_{\mathrm{pre}}$ 且 reason 不是 ok。Index 0 在 \texttt{rank} 里 \texttt{keep} 恒为真，所以到不了 EA / DS。低分即使有部分命中也不走 HP。}
\end{frame}

\section{结果}

\begin{frame}{Prefill pruning 比这三个 host 省在哪里？}
  \why{Loop 的分数约 $0.02$，低于 $0.15$，进 kernel 之前丢掉。策略分数约 $1.0$，留下。Admission 不读 host 的切点，所以三个 host 的 prefill 一起下降。ESC 省得最多，因为它本来要把 loop decode 到窗口结束。}
  {\footnotesize
  \setlength{\tabcolsep}{5pt}
  \begin{tabular}{@{}lrrrr@{}}
    \toprule
    \textbf{$k{=}16$} & \textbf{prefill} & \textbf{e2e，单独} & \textbf{e2e，$+\tau_{\mathrm{pre}}{=}0.15$} & \textbf{any} \\
    \midrule
    ESC & $5008\to 3848$ & 17.05\,s & \good{14.13\,s} & $16/16$ \\
    Speculative Rejection & $5008\to 3848$ & 13.14\,s & \good{12.47\,s} & $16/16$ \\
    DPTS & $5008\to 3848$ & 11.86\,s & \good{11.49\,s} & $16/16$ \\
    \bottomrule
  \end{tabular}}
  \src{同一 JSON。$k{=}8$ 上 SR 和 DPTS 单独跑是 $15/16$，加上 $0.15$ 后是 $16/16$。ESC 的 decode token 从 131072 降到 98304。}

\begin{frame}{四个问题，各用一组数回答}
  \begin{enumerate}
    \item \textbf{为什么三篇的 prefill 一样？} 分数来自已经 resident 的 token。$k{=}16$ 时 ESC、SR、DPTS 的 prefill 都是 5008。加上 $\tau_{\mathrm{pre}}{=}0.15$ 后是 3848，ESC 从 17.05\,s 到 14.13\,s，仍是 $16/16$。
    \vspace{0.18em}
    \item \textbf{为什么不能共用一条 threshold？} Residual 上 illegal 是 $0.22$，生成前缀上仍是 illegal 是 $0.16$。MATH-500 $k{=}16$ 只用 $0.45$ 做 prefill 是 $13/32$、$52.1$\,s；$(0.15,0.45)$ 回到 $20/32$、$72.2$\,s。Full fan-out 是 $20/32$、$104.1$\,s。
    \vspace{0.18em}
    % \item \textbf{为什么默认是 $0.15$？} $k{=}16$ 上 $0.45$ 更快（ESC 10.92\,s，仍是 $16/16$），但 $k{=}4$ 和 $k{=}8$ 掉到 $14/16$。
    \item \textbf{我们的方法把效率提高了多少？} Game of 24 上 4B $1.61\to 1.13$\,s（$30\%$），8B $2.15\to 1.52$\,s（$29\%$）。GSM8K 8B 三轮 $2.48\to 1.91$\,s（$23\%$），只 prefill $48/192$。MATH-500 $k{=}16$ 的 probe $104.1\to 72.2$\,s（$31\%$），$20/32$ 不变。
    \vspace{0.18em}
    \item \textbf{Decoding pruning、prefill pruning、合并？} Decoding pruning 不设 $\tau_{\mathrm{pre}}$，prefill 仍是 5008：ESC 17.05\,s，SR 13.14\,s，DPTS 11.86\,s。Prefill pruning（ESC，$\tau{=}0.15$）是 14.13\,s。合并的 DPTS$+0.15$ 是 11.49\,s。三轮 ToT 没接这三种算法：先 \texttt{queue\_fanout\_prefills}，live 孩子再 decode 32 token，只 prefill $48/192$。
  \end{enumerate}
\end{frame}
